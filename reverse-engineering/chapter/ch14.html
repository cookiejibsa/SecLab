Frida 런타임 후킹

JavaScript로 실행 중인 프로세스에 코드를 인젝션해 행위를 조작한다

Frida의 동작 원리

Frida는 타겟 프로세스에 동적 라이브러리를 인젝션하고, V8 JavaScript 엔진을 심는다. 이 엔진을 통해 사용자가 작성한 JS 코드가 타겟의 메모리 공간에서 실행된다. 함수 후킹, 메모리 읽기/쓰기, 함수 직접 호출이 모두 가능하다.

기본 API

// 함수 후킹 — 진입/반환 시 콜백
Interceptor.attach(targetPtr, {
    onEnter: function(args) {
        // args[0], args[1]... = 함수 인자
        console.log('arg0: ' + args[0]);
        console.log('arg0 as string: ' + args[0].readUtf8String());
    },
    onLeave: function(retval) {
        // 반환값 조작
        retval.replace(0);   // 반환값을 0으로 변경
    }
});

// 함수 교체 (원본 호출 없음)
Interceptor.replace(targetPtr, new NativeCallback(
    function(arg0) {
        console.log('replaced!');
        return 1;
    }, 'int', ['pointer']
));
// 메모리 읽기
var buf = Memory.readByteArray(ptr('0x401000'), 64);
console.log(hexdump(buf));

var str = Memory.readUtf8String(ptr('0x401234'));
var num = Memory.readU32(ptr('0x401234'));

// 메모리 쓰기
Memory.writeByteArray(ptr('0x401000'), [0x90, 0x90]);  // NOP 패치
Memory.writeUtf8String(ptr('0x402000'), 'patched');

// 메모리 검색
Memory.scan(ptr('0x400000'), 0x10000, '4D 5A ?? ?? 00 00', {
    onMatch: function(addr, size) { console.log('Found at: ' + addr); },
    onComplete: function() {}
});
// 모듈 정보
Process.enumerateModules().forEach(function(m) {
    console.log(m.name + ' @ ' + m.base);
});

// 함수 주소 찾기
var addr1 = Module.findExportByName('kernel32.dll', 'CreateFileA');
var addr2 = Module.findExportByName(null, 'malloc');  // 모든 모듈 검색

// 모듈 기준 오프셋으로 주소 계산
var base = Module.findBaseAddress('target.exe');
var target = base.add(0x1234);  // ASLR 우회

// 수동 Frida 스크립트
var SSLContext = Java.use('javax.net.ssl.SSLContext');
SSLContext.init.overload(
    '[Ljavax.net.ssl.KeyManager;',
    '[Ljavax.net.ssl.TrustManager;',
    'java.security.SecureRandom'
).implementation = function(km, tm, sr) {
    // TrustManager를 모든 인증서를 허용하는 것으로 교체
    this.init(km, [getTrustManager()], sr);
};

frida-trace로 빠른 시작

frida-trace -i "CreateFile*" target.exe로 패턴에 맞는 함수를 자동으로 후킹하고 핸들러 JS 파일을 생성해준다. 처음 타겟을 분석할 때 빠르게 호출 패턴을 파악하는 데 유용하다.
