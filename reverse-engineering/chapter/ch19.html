반드시 격리된 VM에서

  • 스냅샷 촬영 확인
  • FakeNet-NG 또는 네트워크 격리 설정
  • Process Monitor 필터 설정 및 캡처 시작
  • Wireshark 캡처 시작
  • 샘플 실행

  • 네트워크 트래픽 분석

YARA 룰 작성

/* 분석된 악성코드의 탐지 룰 작성 */
rule Malware_XYZ_RAT {
    meta:
        description = "XYZ RAT C2 통신 패턴"
        author = "분석가"
    strings:
        $magic = { DE AD C0 DE }        // C2 패킷 매직
        $ua    = "Mozilla/4.0 (XYZ)"    // 커스텀 UA
        $mutex = "Global\XYZ_Mutex"    // 뮤텍스 이름
    condition:
        uint16(0) == 0x5A4D and         // PE 파일
        ($magic or $ua or $mutex)
}

# YARA로 스캔
yara rule.yar target_directory/
